## Appendix A: `make_exp.py`

Run it from the repository root after changing the two `/home/ubuntu/audit/wt` paths to your checkout.

```python
#!/usr/bin/env python3
"""Make an instrumented local copy of visual.e2e.ts from a given commit.
usage: make_exp.py <commit> <out-name> <phone-loop-timeout-ms> [stop-at-timeout]
The copy differs from the commit's file ONLY by: (1) a log helper + beforeEach/afterEach that log test start/end,
(2) one log line per phone-loop iteration, (3) the phone loop's timeout number,
(4) optional 'stop-at-timeout': the phone loop breaks when Vitest's abort signal fires (control experiment)."""
import subprocess, sys
commit, out, timeout = sys.argv[1], sys.argv[2], sys.argv[3]
stop = len(sys.argv) > 4 and sys.argv[4] == 'stop-at-timeout'
src = subprocess.check_output(['git', 'show', f'{commit}:apps/studio/e2e/visual.e2e.ts'], cwd='/home/ubuntu/audit/wt').decode()
def rep(a, b):
    global src
    assert src.count(a) == 1, (a, src.count(a))
    src = src.replace(a, b)
rep("const require = createRequire(import.meta.url);",
    "import { appendFileSync as __append } from 'node:fs';\nimport { beforeEach as __be, afterEach as __ae } from 'vitest';\n"
    "const __t0 = Date.now();\nconst __log = (s: string): void => __append(process.env['EXP_LOG']!, `${String(Date.now() - __t0).padStart(7)} ms  ${s}\\n`);\n"
    "__be((ctx) => __log(`TEST START  ${ctx.task.name.slice(0, 70)}`));\n__ae((ctx) => __log(`TEST END    ${ctx.task.name.slice(0, 70)}`));\nlet __iter = 0;\n"
    "const require = createRequire(import.meta.url);")
sig = "async ({ signal })" if stop else "async ()"
rep("it('never makes the page wider than a phone, whatever the Direction, the type Options, or the slider ends', async () => {",
    f"it('never makes the page wider than a phone, whatever the Direction, the type Options, or the slider ends', {sig} => {{")
head_call = "        await show(mockup(design.params, 'top'), page);\n        const report = await page.evaluate(() => {"
base_call = "        await show(mockup(design.params, 'top'));\n        const report = await page.evaluate(() => {"
call = head_call if head_call in src else base_call
brk = "        if (signal.aborted) { __log('PHONE LOOP stopped by the experiment at the timeout'); return; }\n" if stop else ""
rep(call, brk + "        __log(`PHONE LOOP iteration ${++__iter}: calls show()`);\n" + call)
rep("  }, 240_000);", f"  }}, {timeout});")
open(f'/home/ubuntu/audit/wt/apps/studio/e2e/{out}.e2e.ts', 'w').write(src)
print('wrote', out)
```
